# 1. Очищаем таблицы структуры
print("Очищаем ClassStructure...")

# 2. Удаляем задачи с id > 59
print("Удаляем задачи с id > 59...")
deleted_count = Task.objects.filter(id__gt=59).delete()
print(f"Удалено задач: {deleted_count}")

# Проверяем результат
print(f"\nОсталось задач: {Task.objects.count()}")
print(f"Максимальный ID: {Task.objects.order_by('-id').first().id if Task.objects.exists() else 'Нет задач'}")
